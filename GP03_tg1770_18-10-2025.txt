
--create tables--

CREATE TABLE User(
  nic CHAR(12) PRIMARY KEY,
  f_name VARCHAR(10),
  l_name VARCHAR(10),
  address VARCHAR(15),
  email VARCHAR (15),
  gender VARCHAR(5),
  bod DATE 
    );

CREATE TABLE Dean(
    dean_id CHAR(5) PRIMARY KEY,
    nic CHAR(12) 
    );

CREATE TABLE Student(   
    student_id VARCHAR(6) PRIMARY KEY, 
    nic CHAR(12),
    state VARCHAR(30),
    dep_id CHAR(4)
    );

CREATE TABLE Attendence(
    date DATE,
    att_state VARCHAR(20),
    session_type VARCHAR(25),
    student_id VARCHAR(6),
    medical_id CHAR(10),
    tech_officer_id VARCHAR(6),
    course_code CHAR(8)
    );

CREATE TABLE User_contact(
    nic CHAR(12) PRIMARY KEY,
    contact_no CHAR(10)
    );
