--------Create Table-------



CREATE TABLE Lecture_student(
    lecture_id VARCHAR(5),
    student_id VARCHAR(6)
    );



CREATE TABLE Admin(   
     nic CHAR(12),
     admin_id VARCHAR(10) PRIMARY KEY,
     role  VARCHAR(10)
    );



CREATE TABLE Lecture(
    lecture_id VARCHAR(5) PRIMARY KEY,
    nic CHAR(12),
    position VARCHAR(20),
    dep_id CHAR(4)
    );

CREATE TABLE Lecture_Course(
    lecture_id VARCHAR(5),
    course_code CHAR(8)
    );
